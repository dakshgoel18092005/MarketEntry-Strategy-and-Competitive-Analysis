SQL Production Query Code: 03_weighted_scoring_model.sql

MySQL 8.0+ Compatible 
WITH normalized_metrics AS (
    SELECT 
        region_id, state_name, grid_region,
        demand_potential, competitive_intensity, expected_profitability,
        ROUND(
            (demand_potential * 0.40) + 
            (expected_profitability * 0.40) + 
            ((100.0 - competitive_intensity) * 0.20), 
            2
        ) AS composite_market_score
    FROM regional_bess_market_data
),
market_rankings AS (
    SELECT 
        region_id, state_name, grid_region,
        composite_market_score,
        RANK() OVER (ORDER BY composite_market_score DESC) AS entry_rank,
        ROUND(composite_market_score - AVG(composite_market_score) OVER (), 2) AS spread_vs_national_avg
    FROM normalized_metrics
)
SELECT entry_rank, state_name, grid_region, composite_market_score, spread_vs_national_avg,
    CASE 
        WHEN entry_rank = 1 THEN 'Wave 1: Immediate Market Entry'
        WHEN entry_rank <= 3 THEN 'Wave 2: Secondary Expansion'
        ELSE 'Wave 3: Opportunistic'
    END AS strategic_recommendation
FROM market_rankings
ORDER BY entry_rank ASC;
